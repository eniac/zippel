Known deltas from the submitted paper (full accounting in README.md):
(1) Completeness count. §9.3 says "Zippel can automatically verify 20 of them (66.7%)"; the corrected count is 18 of 30. Minus 5: the whole HyperPlonk family and Dekart were false positives from a bug that has since been fixed. Plus 3: Groth16, KZH and Bulletproofs (ipa) each had a where clause missing required constraints, now fixed. 20 − 5 + 3 = 18.
(2) LoC drift. Figure 6 and the Zippel LoC column of Figure 7 have moved, for two reasons: zippel-fmt did not exist at submission time and reformatting the examples changed line counts on its own; and several where clauses were revised (see (1)). Native baseline LoC are unchanged — that code is vendored and never formatted or edited here — and Zippel stays shorter than native for all nine benchmarked systems (README's 9-row table).
(3) Figure 7 Hyrax. The submitted prose claimed a prover speedup up to 6.1× (1 thread) and verifier 3.4×. Corrected during review to 1.11× prover at 1 thread and 3.30× verifier; the prover ratio dips between 1 and 2 threads. This is expected and has been discussed with the reviewers.
(4) Recompiles. Each docker run --rm triggers one partial recompile because the container is ephemeral. Mount a volume at /zippel/target to avoid paying it repeatedly.

// Sample a univariate polynomial one point at a time via single-scalar
// application `p(t)`. Binary `eval(p, vector)` on a univariate poly is a type
// error: a one-variable residual would misread the vector as one multivariate
// coordinate point instead of a batch of scalar sample points.
fn sample_univariate<F: Field, MAX_DEGREE_CONST: Size>(
    instance p: Uni<F, MAX_DEGREE_CONST>,
    instance points: [F; MAX_DEGREE_CONST + 1],
) -> [F; MAX_DEGREE_CONST + 1] {
    [p(t) for t in points]
}

// Round 2 onward (prover): check g(0)+g(1) = previous g(r), then recurse on the explicit
// residual polynomial until `Poly<F,1>`.
//
// orig_poly was removed: the final subclaim `eval(orig_poly, full_challenges)`
// for the original-polynomial opening
// is delegated to the prover (matches hyperplonk's SumCheck::verify which
// returns a subclaim and leaves the polynomial opening to the caller).
// With orig_poly gone from the protocol surface, the verifier no longer
// absorbs the full polynomial into the FS transcript at the start (which
// was the dominant O(2^NV) cost in verify) — making verify O(log N)
// instead of O(N).
fn sumcheck_round<F: Field, NUM_VARS_CONST: Size, V: 3..NUM_VARS_CONST + 1, MAX_DEGREE_CONST: Size>(
    witness curr_poly: Poly<F, V, MAX_DEGREE_CONST>,
    instance points: [F; MAX_DEGREE_CONST + 1],
    instance challenges: [F; NUM_VARS_CONST - V + 1],
    instance prev_eval: F,
    instance ch: F,
    instance curr_round: Fin<0..NUM_VARS_CONST>,
) -> Unit {
    let residual_poly = eval(curr_poly, [ch]);
    // Round polynomial: sum the residual evaluated over the Boolean-tail
    // hypercube. This lowers to a generic materialized reduce-map (the 2^k
    // tail domain is materialized; Groebner completeness const-folds the bits).
    let round_poly = reduce(
        +,
        [
            eval<0>(residual_poly, tail)
            for tail in [[i / 2 ^ j % 2 for j in 0..V - 2] for i in 0..2 ^ (V - 2)]
        ],
    );
    let eval_points = points;
    let evs = sample_univariate(round_poly, eval_points);

    evs_msg <- evs;
    verify(prev_eval == evs_msg[0] + evs_msg[1]);

    let g = interpolate(eval_points, evs_msg);
    r_next <- challenge<F>;
    let next_prev = g(r_next);

    let next_challenges = challenges ++ [r_next];
    sumcheck_round(residual_poly, points, next_challenges, next_prev, r_next, curr_round + 1)
}

fn sumcheck_round<F: Field, NUM_VARS_CONST: Size, MAX_DEGREE_CONST: Size>(
    witness curr_poly: Poly<F, 2, MAX_DEGREE_CONST>,
    instance points: [F; MAX_DEGREE_CONST + 1],
    instance challenges: [F; NUM_VARS_CONST - 1],
    instance prev_eval: F,
    instance ch: F,
    instance curr_round: Fin<0..NUM_VARS_CONST>,
) -> Unit {
    let residual_poly = eval(curr_poly, [ch]);
    let eval_points = points;
    let evs = sample_univariate(residual_poly, eval_points);
    evs_msg <- evs;
    verify(prev_eval == evs_msg[0] + evs_msg[1]);

    let g = interpolate(eval_points, evs_msg);
    r_final <- challenge<F>;
    let _drop_g = g;
    let _drop_r = r_final;
    let _drop_round = curr_round;
    let _drop_challenges = challenges;
    verify(1 == 1)
}

//
// The `where` clause specifies the trusted setup directly: sample a random
// trapdoor tau, then derive the SRS as powers of tau times the generators.
// This replaces the previous indirect pairing-based SRS check.
//
// `srs_g1` is `extra` even though the SRS is not cryptographically secret
// — the verifier graph never consumes it (only the prover does, for the
// commit + opening MSMs), and zippel's runtime currently requires every
// `instance` input to be passed to the verifier. Marking it `extra` sidesteps
// the unused-input overhead.
proto kzg<G1: Group, G2: Group, GT: Pairing<G1, G2>, F: Scalar<G1, G2>, N: Size>(
    witness poly_x: Uni<F, N - 1>,
    instance eval_point: F,
    instance eval_result: F,
    extra srs_g1: [G1; N],
    instance gen_g1: G1,
    instance gen_g2: G2,
    instance srs_g2_s: G2,
) where
    let tau = random<F>;
    reduce(&&, srs_g1 == [tau ^ i * gen_g1 for i in 0..N])
        && srs_g2_s == tau * gen_g2
        && poly_x(eval_point) == eval_result
{

    let quotient_poly = (poly_x - eval_result) / poly([-eval_point, 1]);

    // The quotient type keeps the dividend's degree bound, but dividing by
    // the exact-degree-1 divisor X - eval_point leaves only N-1 meaningful
    // coefficients; the discarded suffix is zero.
    let quotient_coeffs = coef(quotient_poly);
    let quotient_coeffs_truncated = quotient_coeffs[0..N - 1];
    let srs_g1_truncated = srs_g1[0..N - 1];
    proof <- dot(quotient_coeffs_truncated, srs_g1_truncated);

    let pairing_lhs = pair(proof, srs_g2_s - gen_g2 * eval_point);
    let pairing_rhs = pair(commitment - eval_result * gen_g1, gen_g2);
    verify(pairing_lhs == pairing_rhs)

// A simplified sumcheck protocol specifically for multilinear polynomials.

// application `p(t)`. Binary `eval(p, vector)` on a univariate poly is a type
// error: a one-variable residual would misread the vector as one multivariate
// coordinate point instead of a batch of scalar sample points.
fn sample_univariate<F: Field, MAX_DEGREE_CONST: Size>(
    instance p: Uni<F, MAX_DEGREE_CONST>,
    instance points: [F; MAX_DEGREE_CONST + 1],

}

// Round 2 onward (prover): check g(0)+g(1) = previous g(r), then recurse on the
// explicit residual polynomial until the residual is `Poly<F,2>`.
fn sumcheck_round<F: Field, NUM_VARS: Size, V: 3..NUM_VARS + 1>(
    instance orig_poly: Mle<F, NUM_VARS>,
    instance curr_poly: Mle<F, V>,
    instance challenges: [F; NUM_VARS - V + 1],

    instance ch: F,
    instance curr_round: Fin<0..NUM_VARS>,

    // This lowers to a generic materialized reduce-map.

        +,
        [
            eval<0>(residual_poly, tail)
            for tail in [[i / 2 ^ j % 2 for j in 0..V - 2] for i in 0..2 ^ (V - 2)]
        ],
    );

    let evs = sample_univariate(round_poly, points);

    verify(prev_eval == evs_msg[0] + evs_msg[1]);

    let g = interpolate(points, evs_msg);
    r_next <- challenge<F>;
    let next_prev = g(r_next);

    let next_challenges = challenges ++ [r_next];
    sumcheck_round(orig_poly, residual_poly, next_challenges, next_prev, r_next, curr_round + 1)
}

// round polynomial against the original polynomial at the full challenge point.

    instance orig_poly: Mle<F, NUM_VARS>,
    instance curr_poly: Mle<F, 2>,
    instance challenges: [F; NUM_VARS - 1],

}

// PST13 Multilinear Polynomial Commitment Scheme
// Reference: Papamanthou, Shi, Tamassia 2013 (PST)
//
// Generic over N variables: the polynomial p has 2^N coefficients indexed
// by {0,1}^N in lexicographic order (MSB-first, so variable 1 is the high bit).
//
// Commit: C = MSM(p, ck) = p̃(α) · g_gen, where ck[i] = eq_N(α, i) · g_gen.
//
// Open p̃(z) = y: log N quotient commitments π_1, ..., π_N. The verifier checks
//   e(C − y·g, h) == Σ_i e(π_i, α_H[i] − z_i·h).
// Equivalently (move RHS to LHS):
//   e(C − y·g, h) + Σ_i e(−π_i, α_H[i] − z_i·h) == 0_GT,
// which fuses into ONE multi_miller_loop + ONE final_exponentiation via the
// `dot(VecG1, VecG2) -> GT` arm in the backend — the same trick used by the
// Groth16 and PARI verifiers in this repo.

// =============================================================================
// eq_mle(z) — multilinear-equality basis at z, MSB-first.
//
// Returns the [F; 2^K] vector whose i-th entry is
//     eq(z, i) = Π_j ( bit_j(i) ? z[j] : 1 - z[j] ),
// where bit_j(i) = (i >> (K-1-j)) & 1 — i.e. z[0] is the high bit and is
// peeled first. This matches the recursion in `pst13_open_rounds` (which
// folds on `z_curr[0]` after splitting `p_curr` on its high half) and the
// `ck_n_scalars` construction in the Rust `prover_create_inputs`.
//
// Used in the proto's `where` clause to express the canonical PCS
// evaluation statement:  y == dot(eq_mle(z), p) == p̃(z).
// =============================================================================
fn eq_mle<G1: Group, G2: Group, GT: Pairing<G1, G2>, F: Scalar<G1, G2>>(
    instance z: [F; 1],
) -> [F; 2] {
    [1 - z[0], z[0]]
}

fn eq_mle<G1: Group, G2: Group, GT: Pairing<G1, G2>, F: Scalar<G1, G2>, K: 2..21>(
    instance z: [F; K],
) -> [F; 2 ^ K] {
    let a = z[0]; // first variable = MSB (peeled first)
    let rest = z[1..K];
    let prev = eq_mle(rest); // [F; 2^(K-1)] over the K-1 low-bit variables
    (prev * (1 - a)) ++ (prev * a)
}

// PST.Commit  (generic over M = number of variables)
//   C_p = dot(p, ck_N) = p(alpha) * G

// PST.Open rounds — BASE CASE (1 variable remaining).
// Logs the final quotient commitment and returns the singleton list [π_last].
// =============================================================================
fn pst13_open_rounds<G1: Group, G2: Group, GT: Pairing<G1, G2>, F: Scalar<G1, G2>>(
    witness p_curr: [F; 2],
    instance ck_curr: [G1; 2],
    instance _z_curr: [F; 1],
) -> [G1; 1] {

    // `pi_last` is the transcript-bound name; the verifier reads it from
    // the proof transcript instead of recomputing from the witness `q`.

}

// PST.Open rounds — RECURSIVE CASE (K >= 2 variables remaining).
// Logs π for this round, folds p at z_curr[0], recurses on K-1 variables.
// Returns the full list of π's collected across rounds.
// =============================================================================
fn pst13_open_rounds<G1: Group, G2: Group, GT: Pairing<G1, G2>, F: Scalar<G1, G2>, K: 2..21>(
    witness p_curr: [F; 2 ^ K],
    instance ck_curr: [G1; 2 ^ K],
    instance z_curr: [F; K],

    // ck_{K-1}[i] = ck_curr[i] + ck_curr[i + 2^(K-1)] (Lagrange-basis split).
    let ck_lo = ck_curr[0..2 ^ (K - 1)];
    let ck_hi = ck_curr[2 ^ (K - 1)..2 ^ K];
    let ck_nxt = ck_lo + ck_hi;

    let p_lo = p_curr[0..2 ^ (K - 1)];

    pi_curr <- dot(q, ck_nxt);

    let result = dot(g1_terms, g2_terms);

// - `ipa_wrapper` implements Protocol 1, which reduces the relation <a, b> = c
//   to a recursive argument by scaling the commitment with the inner product value
//   (P' = P + u * (x * c)).
// - `ipa_recursive` implements Protocol 2, the recursive vector-folding argument.
//
// Reference: Section 3 of "Bulletproofs: Short Proofs for Confidential Transactions and More"
// (https://eprint.iacr.org/2017/1066).
// curve: ArkSecp256k1
// N_current: Current dimension of vectors g, h, a, b.
// u_base_for_ipa: The 'u' base to be used for L,R commitments and base case check in this IPA run.
fn ipa_recursive<G: Group, F: Scalar<G>>(
    instance g_vec_k: [G; 1], // Current g bases for this round
    instance h_vec_k: [G; 1], // Current h bases for this round
    instance u_base_for_ipa: G, // The 'u' to use for this IPA instance's L,R, and base check
    instance p_commitment_k: G, // Current P commitment for this round
    witness a_vec_k: [F; 1], // Prover's current 'a' vector (witness)
    witness b_vec_k: [F; 1], // Prover's current 'b' vector (witness)
) -> Unit {
    // BASE CASE (n=1)
    // Prover sends the final folded a, b scalars — these are the proof
    // elements that close out the protocol (BBB+18 Protocol 2 line 8).
    // In the non-interactive setting, they are the final prover results
    // used for the verifier's final check.
    final_a_scalar <- a_vec_k[0];
    final_b_scalar <- b_vec_k[0];

    // Verifier recomputes these locally using its instance bases and the
    // proof-supplied final scalars. Soundness depends on it: ga, hb,
    // and ip_val_at_base/c = a*b are NOT proof elements.
    let ga = g_vec_k[0] * final_a_scalar;
    let hb = h_vec_k[0] * final_b_scalar;
    let ip_val_at_base = final_a_scalar * final_b_scalar;

    // Verifier checks if p_commitment_k == g_vec_k[0]*final_a_scalar + h_vec_k[0]*final_b_scalar + u_base_for_ipa*ip_val_at_base
    // (Translated from multiplicative g^a h^b u^c)
    let expected_P_at_base = ga + hb + u_base_for_ipa * ip_val_at_base;

    verify(p_commitment_k == expected_P_at_base)
}

fn ipa_recursive<G: Group, F: Scalar<G>, S: Size, N_current: 1..S + 1>(
    instance g_vec_k: [G; 2 ^ N_current], // Current g bases for this round
    instance h_vec_k: [G; 2 ^ N_current], // Current h bases for this round
    instance u_base_for_ipa: G, // The 'u' to use for this IPA instance's L,R, and base check
    instance p_commitment_k: G, // Current P commitment for this round

    // Prover computes c_L, c_R and then L, R.

    let h_v_hi = h_vec_k[2 ^ (N_current - 1)..2 ^ N_current];

    let cL_val = dot(a_p_lo, b_p_hi); // Dot product <a_lo, b_hi>
    let cR_val = dot(a_p_hi, b_p_lo); // Dot product <a_hi, b_lo>

    let b_prime = b_p_lo * inverse + b_p_hi * x_challenge;

        h_prime_next,
        u_base_for_ipa, // Pass the same u received by this function call
        p_prime_next,
        a_prime,
        b_prime,
    )
}

    witness y: F, // Value committed in big_y

    witness r_Y: F, // Randomness for big_y
    witness r_Z: F, // Randomness for big_z (such that big_z commits x*y)

    instance g: G, // Base G1 for Pedersen commitments
    instance h: G, // Base G2 for Pedersen commitments
    instance big_x: G, // Commitment big_x = g*x + h*r_X

    instance big_z: G, // Commitment big_z = g*(x*y) + h*r_Z
)

    let b2 = random<F>;

    let b5 = random<F>;

// Proves knowledge of witness w such that Az ∘ Bz = Cz where z = (x, w)
// Uses Pedersen vector commitments over a field: Commit(ck, v; ω) = dot(ck, v) + h*ω
//
// Generic over: N (total variables), n (instance inputs, n < N), m (constraints).
// w ∈ F^(N-n), z ∈ F^N, A/B/C are m×N (stored flat as [F; m*N]).

// Matrix-vector product for an m×N matrix (flat, row-major) times an N-vector.
fn matvec<G: Group, F: Scalar<G>, N: Size, m: Size>(
    instance mat: [F; m * N],
    instance v: [F; N],
) -> [F; m] {
    [reduce(+, [mat[i * N + j] * v[j] for j in 0..N]) for i in 0..m]
}

proto r1cs_sigma<G: Group, F: Scalar<G>, N: Size, n: 1..N, m: Size>(
    // Instance inputs
    instance ck: [G; m], // Commitment key (m generators)
    instance h_base: G, // Blinding base
    instance mat_A: [F; m * N], // Matrix A (m × N, flattened row-major)
    instance mat_B: [F; m * N], // Matrix B
    instance mat_C: [F; m * N], // Matrix C
    instance x: [F; n], // Instance input portion of z

    // Witness inputs
    witness w: [F; N - n], // Witness portion of z
) where reduce(&&, matvec(mat_A, x ++ w) * matvec(mat_B, x ++ w) == matvec(mat_C, x ++ w)) {
    // ======== Prover Phase 1: Compute and commit ========

    // Build full vector z = (x, w)
    let z = x ++ w;

    // Sample random blinding scalar r ∈ F
    let r = random<F>;

    // Compute matrix-vector products: z_X = X·z
    let z_A = matvec(mat_A, z);
    let z_B = matvec(mat_B, z);
    let z_C = matvec(mat_C, z);

    let omega_A = random<F>;
    let omega_B = random<F>;
    let omega_C = random<F>;

    // Build padded randomness (0^n, r) ∈ F^N

    // Compute r_X = X·(0^n, r)
    let r_A = matvec(mat_A, padded_r);
    let r_B = matvec(mat_B, padded_r);
    let r_C = matvec(mat_C, padded_r);

    // Commit to r_A, r_B, r_C with fresh blinding
    let omega_A' = random<F>;
    let omega_B' = random<F>;
    let omega_C' = random<F>;

    // Cross-term commitments
    let omega_1 = random<F>;
    let omega_2 = random<F>;

    // ======== Send commitments (8 values) ========
    t_C_A <- dot(ck, z_A) + h_base * omega_A;
    t_C_B <- dot(ck, z_B) + h_base * omega_B;
    t_C_C <- dot(ck, z_C) + h_base * omega_C;
    t_C_A' <- dot(ck, r_A) + h_base * omega_A';
    t_C_B' <- dot(ck, r_B) + h_base * omega_B';
    t_C_C' <- dot(ck, r_C) + h_base * omega_C';
    t_C_1 <- dot(ck, z_A * r_B + z_B * r_A) + h_base * omega_1;
    t_C_2 <- dot(ck, r_A * r_B) + h_base * omega_2;

    // ======== Challenge ========
    gamma <- challenge<F>;

    // ======== Prover Phase 2: Responses ========

    // s = w + γ·r ∈ F^(N-n)
    t_s <- [w[i] + gamma * r for i in 0..N - n];
    t_sigma_A <- omega_A + gamma * omega_A';
    t_sigma_B <- omega_B + gamma * omega_B';
    t_sigma_C <- omega_C + gamma * omega_C';
    t_sigma_o <- omega_C + gamma * omega_1 + gamma * gamma * omega_2;

    // ======== Verifier checks ========

    // Compute s_X = X·(x, s)
    let s_A = matvec(mat_A, x ++ t_s);
    let s_B = matvec(mat_B, x ++ t_s);
    let s_C = matvec(mat_C, x ++ t_s);

    // Check 1–3: C_X + γ·C'_X == Commit(ck, s_X; σ_X)
    verify(t_C_A + gamma * t_C_A' == dot(ck, s_A) + h_base * t_sigma_A);
    verify(t_C_B + gamma * t_C_B' == dot(ck, s_B) + h_base * t_sigma_B);
    verify(t_C_C + gamma * t_C_C' == dot(ck, s_C) + h_base * t_sigma_C);

    // Check 4: C_C + γ·C_1 + γ²·C_2 == Commit(ck, s_A ∘ s_B; σ_o)

}

fn perm_piop_column<F: Field, S: Size, N: 2 ^ S>(

    let v_lead_1 = mle(v_lead_1_evs);

// Zero-Knowledge KZG polynomial commitment scheme.
//
// Standard KZG leaks information about p(x) through the commitment and proof.
// To achieve ZK, the prover blinds p(x) with random polynomial delta(x)*(x - z)
// where delta(x) = b0 + b1*x. Since the blinding vanishes at x = z,
// we still have p_tilde(z) = p(z) = y, so the evaluation is preserved.
//
// The quotient q_tilde(x) = (p_tilde(x) - y) / (x - z) = q(x) + b0 + b1*x
// is now randomized, and the pairing check remains identical.
//
// SRS needs N+1 elements (one extra) to commit to the degree-N blinded polynomial.
// The `where` clause specifies the trusted setup directly: sample a random
// trapdoor tau, then derive the SRS as powers of tau times the generators.

proto zk_kzg<G1: Group, G2: Group, GT: Pairing<G1, G2>, F: Scalar<G1, G2>, N: Size>(
    witness p_val: Uni<F, N - 1>,
    instance z: F,
    instance y: F,
    extra ss: [G1; N + 1],
    instance g: G1,
    instance h: G2,
    instance h_val: G2,
) where
    let tau = random<F>;
    reduce(&&, ss == [tau ^ i * g for i in 0..N + 1]) && h_val == tau * h && p_val(z) == y
{
    // Sample blinding factors
    let b0 = random<F>;
    let b1 = random<F>;

    // Blinding polynomial: (b0 + b1*x)*(x - z) = -b0*z + (b0 - b1*z)*x + b1*x^2
    let blind_coefs = [-b0 * z, b0 - b1 * z, b1];
    let blind_val = poly(blind_coefs);

    // Blinded polynomial: p_tilde(x) = p(x) + blind(x)
    // Note: p_tilde(z) = p(z) + 0 = y (blinding vanishes at z)
    let p_tilde = p_val + blind_val;
    let p_tilde_coefs = coef(p_tilde);

    // Commit to the blinded polynomial (needs N+1 SRS elements)
    c <- dot(p_tilde_coefs, ss);

    // Quotient: q_tilde(x) = (p_tilde(x) - y) / (x - z) = q(x) + b0 + b1*x
    let q_tilde = (p_tilde - y) / poly([-z, 1]);
    // The quotient keeps p_tilde's declared bound; the monic linear divisor
    // X - z leaves N meaningful coefficients for the proof MSM.
    let q_coefs = coef(q_tilde);
    let q_coefs_truncated = q_coefs[0..N];
    let ss_q = ss[0..N];
    pi <- dot(q_coefs_truncated, ss_q);

    // Verification: same pairing check as standard KZG
    let lhs = pair(pi, h_val - h * z);
    let rhs = pair(c - y * g, h);
    verify(lhs == rhs)
}

// PARI: Faster and Smaller SNARK via Equifficient Polynomial Commitments
// (Dellepere, Mishra, Shirzad — Sec USENIX 2026)
//
// PARI is a SNARK for Square R1CS (SR1CS): given matrices A, B in F^{m x k}
// and an instance/witness split z = (x, w), prove that
//   Az ∘ Az = Bz                  (Hadamard product over m constraints).
//
// This file implements PARI's prover and verifier per Fig. 6 of the paper,
// with the same matrix layout as the reference impl
// (alireza-shirzad/garuda-pari) so that the succinct-verifier Lagrange
// shortcut applies.
//
// Layout assumption (upstream "instance outliner"):
//   - Constraints 0..K-N: A's instance columns are zero (so x̂_A vanishes
//     on those K-N domain points), B's instance columns are zero
//     everywhere (so x̂_B ≡ 0 on K).
//   - Constraints K-N..K (the last N): A row = e_i (e.g. A[K-N+i][i]=1)
//     so that x̂_A(ω^{K-N+i}) = x[i].
// Under this layout, x̂_A is determined by the N instance inputs alone, and
// the verifier can compute x̂_A(r) via the closed-form Lagrange shortcut:
//   x̂_A(r) = (v_K(r) / K) · Σ_{i=0..N} omegas[i] · x[i] / (r - omegas[i])
// where omegas[i] = ω^{K-N+i}.
//
// Prover-side simplification: instead of feeding A, B as sparse matrices
// (no zippel support), the prover precomputes and feeds:
//     z_a_evals := A·z       (length K, witness)
//     z_b_evals := B·z       (length K, witness)
//     w_a_evals := A·(0∥w)   (length K, witness; = z_a_evals − x̂_A on K)
//     w_b_evals := B·(0∥w)   (length K, witness; = z_b_evals under layout)
//
// Parameters:
//   M    : log_2(K) where K is the constraint-domain size.
//   K    : number of constraints (= 2^M). K is a smooth subgroup.
//   N    : number of instance input variables (z[0]=1 plus user inputs).
//   KMN  : k - N where k = total number of variables. The witness w lives in F^KMN.
//
// Soundness in AGM (per the paper); not zero-knowledge in this version
// (omits the hiding/zk-masking that adds dummy random constraints).

fn eq_weights<G: Group, F: Scalar<G>, K: 2..21>(instance x: [F; K]) -> [F; 2 ^ K] {
    let x_lo = x[0..K - 1];

    witness r2: F, // Randomness for second commitment

// This file implements the Inner Product Argument from the Bulletproofs paper
// with pre-weighted/scaled generator bases (often used in range proofs and R1CS batch checks):
// - `ipa_wrapper` implements Protocol 1, which reduces the relation <a, b> = c
//   to a recursive argument by scaling the commitment with the inner product value
//   (P' = P + u * (x * c)) using weighted generator bases.
// - `ipa_recursive` implements Protocol 2, the recursive vector-folding argument.
//
// Reference: Sections 3 & 4 of "Bulletproofs: Short Proofs for Confidential Transactions and More"
// (https://eprint.iacr.org/2017/1066).
// curve: ArkSecp256k1
// N_current: Current dimension of vectors g, h, a, b.
// u_base_for_ipa: The 'u' base to be used for L,R commitments and base case check in this IPA run.
fn ipa_recursive<G: Group, F: Scalar<G>>(
    instance g_vec_k: [G; 1], // Current g bases for this round
    instance h_vec_k: [G; 1], // Current h bases for this round
    instance u_base_for_ipa: G, // The 'u' to use for this IPA instance's L,R, and base check
    instance p_commitment_k: G, // Current P commitment for this round
    witness a_vec_k: [F; 1], // Prover's current 'a' vector (witness)
    witness b_vec_k: [F; 1], // Prover's current 'b' vector (witness)
) -> Unit {
    // BASE CASE (n=1)
    // Prover "sends" a and b scalars. In non-interactive, these are the final results
    // of the prover's computation, used for the verifier's final check.
    final_a_scalar <- a_vec_k[0];
    final_b_scalar <- b_vec_k[0];

    // Verifier recomputes these locally using its instance bases and the
    // proof-supplied final scalars. Soundness depends on it: ga, hb,
    // and ip_val_at_base are NOT proof elements.
    let ga = g_vec_k[0] * final_a_scalar;
    let hb = h_vec_k[0] * final_b_scalar;
    let ip_val_at_base = final_a_scalar * final_b_scalar;

    // Verifier checks if p_commitment_k == g_vec_k[0]*final_a_scalar + h_vec_k[0]*final_b_scalar + u_base_for_ipa*ip_val_at_base
    // (Translated from multiplicative g^a h^b u^c)
    let expected_P_at_base = ga + hb + u_base_for_ipa * ip_val_at_base;

    verify(p_commitment_k == expected_P_at_base)
}

fn ipa_recursive<G: Group, F: Scalar<G>, S: Size, N_current: 1..S + 1>(
    instance g_vec_k: [G; 2 ^ N_current], // Current g bases for this round
    instance h_vec_k: [G; 2 ^ N_current], // Current h bases for this round
    instance u_base_for_ipa: G, // The 'u' to use for this IPA instance's L,R, and base check
    instance p_commitment_k: G, // Current P commitment for this round
    witness a_vec_k: [F; 2 ^ N_current], // Prover's current 'a' vector (witness)
    witness b_vec_k: [F; 2 ^ N_current], // Prover's current 'b' vector (witness)
) -> Unit {
    // RECURSIVE STEP (n > 1)

    // Prover computes c_L, c_R and then L, R.
    // Slice prover's witness vectors a and b
    let a_p_lo = a_vec_k[0..2 ^ (N_current - 1)];
    let a_p_hi = a_vec_k[2 ^ (N_current - 1)..2 ^ N_current];
    let b_p_lo = b_vec_k[0..2 ^ (N_current - 1)];
    let b_p_hi = b_vec_k[2 ^ (N_current - 1)..2 ^ N_current];

    // Slice current g and h base vectors (these are instance for this round)
    let g_v_lo = g_vec_k[0..2 ^ (N_current - 1)];
    let g_v_hi = g_vec_k[2 ^ (N_current - 1)..2 ^ N_current];
    let h_v_lo = h_vec_k[0..2 ^ (N_current - 1)];
    let h_v_hi = h_vec_k[2 ^ (N_current - 1)..2 ^ N_current];

    let cL_val = dot(a_p_lo, b_p_hi); // Dot product <a_lo, b_hi>
    let cR_val = dot(a_p_hi, b_p_lo); // Dot product <a_hi, b_lo>

    // Prover "sends" L and R by adding them to the transcript.
    // L = (g_hi^a_lo) * (h_lo^b_hi) * (u^c_L)  (multiplicative notation)
    // L = g_hi*a_lo) + reduce(+, h_lo*b_hi) + u_base_for_ipa*c_L (additive)
    l_transcript <- dot(g_v_hi, a_p_lo) + dot(h_v_lo, b_p_hi) + u_base_for_ipa * cL_val;
    r_transcript <- dot(g_v_lo, a_p_hi) + dot(h_v_hi, b_p_lo) + u_base_for_ipa * cR_val;

    // Verifier "sends" challenge x (derived from transcript using Fiat-Shamir)
    x_challenge <- challenge<F>;

    let inverse = 1 / x_challenge;
    let x_sq = x_challenge * x_challenge;
    let inv_sq = inverse * inverse;
    // Both Prover and Verifier compute parameters for the next recursive call:
    // g' = g_lo^(x^-1) * g_hi^x  (multiplicative)
    // g_prime = g_lo * inverse + g_hi * x_challenge (additive)
    let g_prime_next = g_v_lo * inverse + g_v_hi * x_challenge;
    let h_prime_next = h_v_lo * x_challenge + h_v_hi * inverse;

    // P' = L^(x^2) * P * R^(x^-2) (multiplicative)
    // p_prime = l_transcript * x_sq + p_commitment_k + r_transcript * inv_sq (additive)
    let p_prime_next = l_transcript * x_sq + p_commitment_k + r_transcript * inv_sq;

    // Prover computes a' and b' for its witness input to the next recursive call
    let a_prime = a_p_lo * x_challenge + a_p_hi * inverse;
    let b_prime = b_p_lo * inverse + b_p_hi * x_challenge;

    // Recursively call protocol_2_ipa_recursive for the next smaller problem instance.
    // The N_current for the recursive call is (N_current / 2).
    // The u_base_for_ipa remains the same throughout this IPA instance.
    ipa_recursive(
        g_prime_next,
        h_prime_next,
        u_base_for_ipa, // Pass the same u received by this function call
        p_prime_next,
        a_prime,
        b_prime,
    )
}

proto ipa_wrapper<G: Group, F: Scalar<G>, S: Size, N_val_const: 2 ^ S>(
    // --- Instance Inputs ---
    instance g_vec: [G; N_val_const], // Corresponds to 'g' in the paper (vector of group elements)
    instance h_vec: [G; N_val_const], // Corresponds to 'h' in the paper (vector of group elements)
    // Generator weighting factors (g_factors, h_factors in the Rust implementation)
    instance g_factors: [F; N_val_const],
    instance h_factors: [F; N_val_const],
    instance p_initial_commitment: G, // Corresponds to 'P' in the paper
    instance ip_val_claimed: F, // Corresponds to 'c' (the inner product value) in the paper
    instance u_aux_base: G, // Corresponds to 'u' in the paper

    // --- Witness Inputs ---
    witness a_vec_witness: [
        F;
        N_val_const
    ], // Corresponds to 'a' in the paper (vector of field elements)
    witness b_vec_witness: [
        F;
        N_val_const
    ], // Corresponds to 'b' in the paper (vector of field elements)
) where
    p_initial_commitment
        == dot(g_vec * g_factors, a_vec_witness) + dot(h_vec * h_factors, b_vec_witness)
        && ip_val_claimed == dot(a_vec_witness, b_vec_witness)
{
    // 1. "Verifier generates x <-$ Z*_p and sends it to Prover."
    //    In our non-interactive DSL, this is a Fiat-Shamir challenge.
    //    The challenge 'x_chal' is a scalar of type F.
    x_chal <- challenge<F>;

    // 2. "Let P' = P * u^(x*c)"
    //    Translating to additive notation: p_prime = p_initial_commitment + u_aux_base * (x_chal * ip_val_claimed)
    //    'x_chal * ip_val_claimed' is a scalar multiplication.
    //    'u_aux_base * (scalar)' is scalar multiplication of a group element.
    let p_prime = p_initial_commitment + u_aux_base * (x_chal * ip_val_claimed);

    //    We also need u^x for the input to Protocol 2.
    //    Translating to additive notation: u_raised_to_x = u_aux_base * x_chal
    let u_raised_to_x = u_aux_base * x_chal;

    // 3. "Run Protocol 2 (ipa recursive) on Input(g, h, u^x, P'; a, b)"
    //    We call the assumed 'protocol_2' function.
    //    The inputs g_vec, h_vec are the original instance base vectors.
    //    u_raised_to_x and p_prime are the values computed above.
    //    a_vec_witness and b_vec_witness are the original witness vectors.
    // Apply generator weighting once at the start (g_factors, h_factors) to obtain
    // the effective bases G', H' used by the recursive IPA, analogous to the
    // "H' = H * y_inv" / "g_factors, h_factors" trick in the Rust code.
    let g_weighted = g_vec * g_factors;
    let h_weighted = h_vec * h_factors;

    // This Protocol 1 succeeds if the called ipa_recursive verifies successfully.
    ipa_recursive(g_weighted, h_weighted, u_raised_to_x, p_prime, a_vec_witness, b_vec_witness)
}

// Zeromorph KZG protocol by Tohru Kohrita and Patrick Towa
// This makes makes KZG Zero-Knowledge. This is not the
// full Multilinear PCS Zeromorph protocol.
// Protocol is from: https://eprint.iacr.org/2023/917.pdf.
//
// The `where` clause specifies the trusted setup directly: sample random
// trapdoors tau and xi, then derive the SRS. srs_g1 holds powers of tau
// (indices 0..N-1) plus xi*gen_g1 at index N. srs_g2 holds tau*gen_g2
// and xi*gen_g2.

proto zeromorph_kzg<G1: Group, G2: Group, GT: Pairing<G1, G2>, F: Scalar<G1, G2>, N: Size>(
    witness p_poly: Uni<F, N - 1>,
    instance u: F,
    instance v: F,
    extra srs_g1: [G1; N + 1],
    instance gen_g1: G1,
    instance gen_g2: G2,
    instance srs_g2: [G2; 2],
) where
    let tau = random<F>;
    let xi = random<F>;
    reduce(&&, srs_g1 == [tau ^ i * gen_g1 for i in 0..N] ++ [xi * gen_g1])
        && reduce(&&, srs_g2 == [tau * gen_g2, xi * gen_g2])
        && p_poly(u) == v
{
    // Sample random blinding factor r
    let r = random<F>;

    // C = [p(tau)]_1 + r[xi]_1
    // srs_g1 contains [1]_1, [tau]_1, ..., [tau^{N-1}]_1, [xi]_1
    let blinded_coeffs = coef(p_poly) ++ [r];
    c <- dot(blinded_coeffs, srs_g1);

    // q(X) = (p(X) - v) / (X - u)
    let q_poly = (p_poly - v) / poly([-u, 1]);

    // Sample random blinding factor s
    let s = random<F>;

    // pi = [q(tau)]_1 + s[xi]_1
    // q's declared bound is the dividend's; the monic linear divisor X - u
    // leaves N-1 meaningful coefficients.
    let q_coeffs = coef(q_poly);
    let q_coeffs_truncated = q_coeffs[0..N - 1];
    let pi_coeffs = q_coeffs_truncated ++ [s];

    let srs_for_pi = srs_g1[0..N - 1] ++ [srs_g1[N]];
    pi <- dot(pi_coeffs, srs_for_pi);

    // delta = r[1]_1 - s[tau]_1 + (s*u)[1]_1

    // srs_g2[0] = [tau]_2, srs_g2[1] = [xi]_2

    let rhs_part1 = pair(pi, srs_g2[0] - u * gen_g2);
    let rhs_part2 = pair(delta, srs_g2[1]);

    verify(lhs == rhs_part1 + rhs_part2)

// Sumcheck with final polynomial opening check.
// Unlike sumcheck.zippel (which delegates the final opening to the prover
// for O(log N) verify), this version threads `orig_poly` through all rounds
// and checks `g(r_final) == eval(orig_poly, full_challenges)` at the end.
// This requires the verifier to see the original polynomial (instance input).

// Sample a univariate polynomial one point at a time via single-scalar
// application `p(t)`. Binary `eval(p, vector)` on a univariate poly is a type
// error: a one-variable residual would misread the vector as one multivariate
// coordinate point instead of a batch of scalar sample points.
fn sample_univariate<F: Field, MAX_DEGREE_CONST: Size>(
    instance p: Uni<F, MAX_DEGREE_CONST>,
    instance points: [F; MAX_DEGREE_CONST + 1],
) -> [F; MAX_DEGREE_CONST + 1] {
    [p(t) for t in points]
}

// Round 2 onward (prover): check g(0)+g(1) = previous g(r), then recurse on the explicit
// residual polynomial until `Poly<F,1>`.
fn sumcheck_round<F: Field, NUM_VARS_CONST: Size, V: 3..NUM_VARS_CONST + 1, MAX_DEGREE_CONST: Size>(
    instance orig_poly: Poly<F, NUM_VARS_CONST, MAX_DEGREE_CONST>,
    witness curr_poly: Poly<F, V, MAX_DEGREE_CONST>,
    instance points: [F; MAX_DEGREE_CONST + 1],
    instance challenges: [F; NUM_VARS_CONST - V + 1],
    instance prev_eval: F,
    instance ch: F,
    instance curr_round: Fin<0..NUM_VARS_CONST>,
) -> Unit {
    let residual_poly = eval(curr_poly, [ch]);
    // Round polynomial: sum the residual evaluated over the Boolean-tail
    // hypercube. This lowers to a generic materialized reduce-map (the 2^k
    // tail domain is materialized; Groebner completeness const-folds the bits).
    let round_poly = reduce(
        +,
        [
            eval<0>(residual_poly, tail)
            for tail in [[i / 2 ^ j % 2 for j in 0..V - 2] for i in 0..2 ^ (V - 2)]
        ],
    );
    let eval_points = points;
    let evs = sample_univariate(round_poly, eval_points);

    evs_msg <- evs;
    verify(prev_eval == evs_msg[0] + evs_msg[1]);

    let g = interpolate(eval_points, evs_msg);
    r_next <- challenge<F>;
    let next_prev = g(r_next);

    let next_challenges = challenges ++ [r_next];
    sumcheck_round(
        orig_poly,
        residual_poly,
        points,
        next_challenges,
        next_prev,
        r_next,
        curr_round + 1,
    )
}

fn sumcheck_round<F: Field, NUM_VARS_CONST: Size, MAX_DEGREE_CONST: Size>(
    instance orig_poly: Poly<F, NUM_VARS_CONST, MAX_DEGREE_CONST>,
    witness curr_poly: Poly<F, 2, MAX_DEGREE_CONST>,
    instance points: [F; MAX_DEGREE_CONST + 1],
    instance challenges: [F; NUM_VARS_CONST - 1],
    instance prev_eval: F,
    instance ch: F,
    instance curr_round: Fin<0..NUM_VARS_CONST>,
) -> Unit {

    let eval_points = points;
    let evs = sample_univariate(residual_poly, eval_points);

    let g = interpolate(eval_points, evs_msg);

    // polynomial evaluated at the full challenge point.
    let full_challenges = challenges ++ [r_final];
    let from_orig = eval(orig_poly, full_challenges);

    let _drop_round = curr_round;
    verify(from_round == from_orig)
}
